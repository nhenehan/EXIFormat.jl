Julia structs

In this chapter you'll describe a document once as Julia structs, then encode and decode it with one call each.

Writing events by hand gets long. EXIFormat.to_exi and EXIFormat.from_exi do the walking for you:

# Names follow the XML: `exi_rename_all` and `exi_names` map Julia field names
# to them.
struct Customer
    loyalty_id::Union{Nothing,String}
    name::String
    tier::String
end
EXIFormat.exi_rename_all(::Type{Customer}) = :PascalCase
EXIFormat.exi_names(::Type{Customer}) = (loyalty_id = "loyaltyId",)

struct Item
    sku::String
    description::String
    quantity::Int32
    price::Float64
end
EXIFormat.exi_rename_all(::Type{Item}) = :PascalCase
EXIFormat.exi_names(::Type{Item}) = (sku = "sku",)

struct Order
    order_id::String
    priority::Union{Nothing,Bool}
    customer::Customer
    items::Vector{Item}
    notes::Union{Nothing,String}
end
EXIFormat.exi_names(::Type{Order}) = (order_id = "orderId", customer = "Customer", items = "Item", notes = "Notes")

The schema decides where each field goes, so you never mark a field as an attribute or an element yourself:

The full list of rules is in the EXIFormat.to_exi docstring (?EXIFormat.to_exi). Encoding and decoding are then one call each:

# Names follow the XML: `exi_rename_all` and `exi_names` map Julia field names
# to them.
struct Customer
    loyalty_id::Union{Nothing,String}
    name::String
    tier::String
end
EXIFormat.exi_rename_all(::Type{Customer}) = :PascalCase
EXIFormat.exi_names(::Type{Customer}) = (loyalty_id = "loyaltyId",)

struct Item
    sku::String
    description::String
    quantity::Int32
    price::Float64
end
EXIFormat.exi_rename_all(::Type{Item}) = :PascalCase
EXIFormat.exi_names(::Type{Item}) = (sku = "sku",)

struct Order
    order_id::String
    priority::Union{Nothing,Bool}
    customer::Customer
    items::Vector{Item}
    notes::Union{Nothing,String}
end
EXIFormat.exi_names(::Type{Order}) = (order_id = "orderId", customer = "Customer", items = "Item", notes = "Notes")


tables = EXIFormat.compile_files(data("order.xsd"))
options = EXIFormat.strict_options()

order = Order("ORD-2026-0007", true, Customer("L-8821", "Raven Bookstore", "gold"),
    [Item("BK-001", "Distributed Systems Patterns", 2, 42.95),
     Item("BK-014", "Practical EXI", 1, 31.50)],
    "gift-wrap requested")

# The root element's name comes from the struct's name, `Order`.
exi = EXIFormat.to_exi(order, tables; options)
from_xml = EXIFormat.encode_xml(read(data("order.xml"), String), tables; options)
println("EXI: ", length(exi), " bytes")
println("same bytes as encoding order.xml: ", exi == from_xml)

back = EXIFormat.from_exi(Order, exi, tables; options)
again = EXIFormat.to_exi(back, tables; options)
println("decodes and encodes to the same bytes: ", again == exi)

println("order ", back.order_id, " for ", back.customer.name, " (", back.customer.tier, ")")
total = 0.0
for item in back.items
    @printf("  %d x %-30s %7.2f\n", item.quantity, item.description, item.price)
    global total += item.quantity * item.price
end
@printf("  total %.2f\n", total)
back.notes === nothing || println("  notes: ", back.notes)
EXI: 129 bytes
same bytes as encoding order.xml: true
decodes and encodes to the same bytes: true
order ORD-2026-0007 for Raven Bookstore (gold)
  2 x Distributed Systems Patterns     42.95
  1 x Practical EXI                    31.50
  total 117.40
  notes: gift-wrap requested

The bytes are the same as encoding the XML file. EXI does not care how you made the events, only what they are.
